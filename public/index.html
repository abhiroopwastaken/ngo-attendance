Option 1: Google Apps Script Webhook (Recommended)

Fastest setup directly from your Google Sheet.

⚠️ Why Webhooks fail & 30-Second Fix
Why you got the error earlier:
  • Who has access: Google defaults to "Only myself". This completely blocks incoming pings. It MUST be set to "Anyone".
  • New Version required: Saving the script does NOT update the web app URL. You must click Deploy → New deployment (or Manage deployments → Edit → New version).
  1. Open your sheet: Click here to open Google Sheet.
  2. In the top menu, click Extensions → Apps Script.
  3. Delete everything in Code.gs and paste the exact script below:
function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({ status: "active", message: "Attendance webhook is online and ready!" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    
    // Auto-create headers if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Record ID", "Timestamp", "Date & Time", "Punch Type", "Worker Name", 
        "Worker ID", "Purpose / Activity", "Site / Location", "Readable Address", 
        "Supervisor Name", "Supervisor Contact", "Latitude", "Longitude", 
        "Accuracy (m)", "Google Maps Link", "Photo URL", "Notes"
      ]);
      sheet.getRange(1, 1, 1, 17).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.id,
      data.timestamp,
      data.formatted_time,
      data.entry_type || "CHECK-IN",
      data.worker_name,
      data.worker_id,
      data.purpose,
      data.site_name,
      data.address || "",
      data.supervisor_name,
      data.supervisor_contact || "",
      data.latitude || "",
      data.longitude || "",
      data.accuracy || "",
      data.maps_url || "",
      data.photo_url || "",
      data.notes || ""
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Click the 💾 Save Project icon (or press Cmd + S / Ctrl + S). (Google will not deploy un-saved code!)
  2. Click Deploy → Manage deployments → click the ✏️ Edit (Pencil) icon → under Version, select New version → click Deploy.
  3. (Or click Deploy → New deployment → Web app → Execute as: Me → Who has access: Anyone).
  4. Copy the resulting Web App URL, paste it above, and click Test Webhook Connection!
